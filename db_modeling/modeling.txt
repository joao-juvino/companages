// platform: dbdiagram.io

Table users {
  id integer [primary key]
  name varchar
  email varchar
  password varchar
  created_at timestamp
}

Table organization {
  id integer [primary key]
  name varchar
  user_id integer
  created_at timestamp
}

Table member {
  id integer [primary key]
  organization_id integer
  created_at timestamp
}

table cargo {
  id integer [primary key]
  name varchar
  weight integer
  organization_id integer
  member_id integer
}


Ref: organization.user_id > users.id // many-to-one

Ref: organization.id < member.organization_id

Ref: organization.id < cargo.organization_id

Ref: member.id < cargo.member_id
